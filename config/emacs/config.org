** Keybindings

*** My Text Editing Rules
#+begin_comment
1. Utilize =meow-*-of-thing= commands for specific selection.
2. For manual selection use =set-mark-command= + movement or =flash-jump= for wider regions.
3. Delete > Kill, because *kill* occupies the =kill-ring=.
4. Utilize =meow-grab= and =meow-swap=
   * In replacement of kill and yank.
   * In replacement of move text in traffic.
   * Use even if it's in another buffer.
5. Use =sync-grab= for duplicating/replace a beacon selection to another.
#+end_comment

*** Helper Functions
#+begin_src elisp
  (defun z/jk-insert-exit()
    "Switch to NORMAL state by quickly typing jk."
    (interactive)
    (let ((event (read-event nil nil 0.2)))
      (if (equal event ?k) (meow-insert-exit)
        (progn (insert "j") (when event (push event unread-command-events)))))
    (message "Back to NORMAL state"))

  (defun my-word-bounds() (bounds-of-thing-at-point 'word))
  (defun z/act-on-thing(fn)
    "Apply FN to region/word/char."
    (cond ((use-region-p) (funcall fn (region-beginning) (region-end)))
  	((my-word-bounds)
  	 (let ((b (car (my-word-bounds)))
  	       (e (cdr (my-word-bounds))))
  	   (funcall fn b e)))
  	((char-after) (funcall fn (point) (1+ (point))))))
#+end_src
*** Smart Helpers for NORMAL state
#+begin_src elisp
  ;; Smart helpers
  (defun z/smart-delete()
    "Delete active region, word, or char at point."
    (interactive) (z/act-on-thing #'delete-region))
  (defun z/smart-change()
    "Change active region, word, or char at point."
    (interactive) (z/smart-delete) (meow-insert))
  (defun z/smart-yank()
    "Yank active region, word, or char at point."
    (interactive) (z/act-on-thing #'copy-region-as-kill))
  (defun z/smart-kill()
    "Kill active region, word, or char at point."
    (interactive) (z/smart-delete) (yank)) ; (z/act-on-thing) #'meow-replace
  (defun z/smart-replace()
    "Replace active region, word, or char at point with yank."
    (interactive) (z/smart-delete) (yank)) ; (z/act-on-thing #'meow-replace
#+end_src

  (defun z/delete-comment(arg)
    "Delete the first comment on this line, if any."
    (interactive "p")
    (let ((kill-ring kill-ring)) (kill-comment arg))
    (message "Comment deleted!"))

  ;; For Leader Keys
  (defun z/kill-current-buffer()
    "Kill current active buffer dwim."
    (interactive)
    (kill-buffer (current-buffer))
    (message "Current buffer killed!"))
